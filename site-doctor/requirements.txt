# Runtime dependencies for Site Doctor.
#
# Trimmed to what the code actually imports. Verified by auditing every
# import statement across agent/ audit/ crawler/ fix/ models/ report/
# secuirty/ triage/ ux_review/ -- including imports inside function bodies,
# not just at the top of files.
#
# The previous version listed 38 packages. 30 were never imported anywhere,
# and their overlapping ">=" constraints made pip's resolver backtrack badly:
# a single docker build downloaded opentelemetry-proto 12 times before being
# abandoned.

beautifulsoup4     # crawler/utils.py           -> bs4
dnspython          # secuirty/verification.py   -> dns.resolver
langgraph          # agent/graph.py             -> StateGraph, END
openai             # triage/, fix/, ux_review/  -> OpenAI client
playwright         # crawler/                   -> sync_playwright
pydantic>=2        # models/schemas.py          -> BaseModel, Field
python-dotenv      # triage/, fix/, ux_review/  -> load_dotenv
reportlab          # report/generate_pdf.py     -> platypus, lib
